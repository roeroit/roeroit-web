function cambia(delta) {
  quantita = Math.max(1, quantita + delta);          // mai sotto 1

JavaScript
document.getElementById("modulo").addEventListener("submit", function (evento) {
  let nome = document.getElementById("nome").value.trim();
  let email = document.getElementById("email").value.trim();
  let errore = document.getElementById("errore");

  if (nome === "") {
    evento.preventDefault();                  // blocca l'invio
    errore.textContent = "Il nome è obbligatorio";
    return;
  }
  if (!email.includes("@") || !email.includes(".")) {
    evento.preventDefault();
    errore.textContent = "Email non valida";
    return;
  }
  errore.textContent = "";                    // tutto ok: il modulo parte
});

Scorciatoia HTML: molte verifiche si fanno anche senza JavaScript, con attributi come required, type="email" e minlength="3". Usa JavaScript quando vuoi messaggi personalizzati o regole speciali.
